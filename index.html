async function deobfuscate() { const raw = inputArea.value.trim(); const zipFile = zipInput.files[0]; if (!raw) { setStatus('Pega un dump primero.', 'error'); return; } if (!zipFile) { setStatus('Sube el .zip del motor.', 'error'); return; } btnDeobf.disabled = true; btnCopy.disabled = true; btnDownload.disabled = true; outputArea.value = ''; setStatus('Procesando VM...', 'loading'); try { const formData = new FormData(); formData.append('raw_text', raw); formData.append('engine_zip', zipFile); const response = await fetch('/api/deobfuscate', { method: 'POST', body: formData }); // Leer como texto primero para evitar el error de JSON const textData = await response.text(); let data; try { data = JSON.parse(textData); } catch (e) { if (textData.includes("The page c")) { throw new Error("Vercel 404: Revisa el archivo vercel.json."); } throw new Error("Respuesta inesperada del servidor (No es JSON)."); } if (!response.ok) { throw new Error(data.detail || 'Error desconocido'); } outputArea.value = data.code; setStatus('¡Completado!', 'success'); btnCopy.disabled = false; btnDownload.disabled = false; } catch (err) { setStatus('Fallo', 'error'); outputArea.value = "[-] ERROR:\n" + err.message; } finally { btnDeobf.disabled = false; } }
